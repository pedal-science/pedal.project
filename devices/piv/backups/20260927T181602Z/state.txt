==> /etc/security/limits.d/90-pedal-rt.conf <==
# Realtime limits for the audio group. Without them a normal user is refused SCHED_FIFO and
# mlockall: tib's defaults are rtprio 0 and an 8 MB memlock. A systemd service does not get
# these (no pam_limits); namchaind's unit must set LimitRTPRIO and LimitMEMLOCK itself.
@audio   -  rtprio    95
@audio   -  memlock   unlimited
# A backstop against a runaway SCHED_FIFO loop, not a latency setting: SIGXCPU after 200 ms of
# realtime CPU without blocking, ~150 periods at 64 frames. Deliberately NOT
# sched_rt_runtime_us=-1, which removes the kernel's only escape from such a loop.
@audio   -  rttime    200000
